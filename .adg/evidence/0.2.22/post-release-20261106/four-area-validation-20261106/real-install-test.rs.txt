use std::path::{Path,PathBuf};
use opencodex_desktop_lib::modules::runtime::{RuntimeStore,RuntimeSourceKind,install::{RuntimeInstaller,SystemNpmRunner,RealVersionProbe,VersionProbe,NullProgressSink,InstallRequest,InstallSource}};
fn tree(p:&Path)->Vec<(String,Vec<u8>)>{fn scan(r:&Path,p:&Path,v:&mut Vec<(String,Vec<u8>)>){for e in std::fs::read_dir(p).unwrap(){let x=e.unwrap().path();if x.is_dir(){scan(r,&x,v)}else{v.push((x.strip_prefix(r).unwrap().to_string_lossy().to_string(),std::fs::read(x).unwrap()))}}}let mut v=vec![];scan(p,p,&mut v);v.sort();v}
#[test]
fn real_archive_real_npm_real_version_recovery() {
 let root=PathBuf::from(std::env::var_os("POST022_TEST_ROOT").unwrap());let data=root.join("actual-installer-isolated-home/data");let home=root.join("actual-installer-isolated-home/home");std::fs::create_dir_all(&home).unwrap();let archive=PathBuf::from("/private/tmp/post021-adopted-runtime-release-20261106/opencodex-2.97.0-post021.ownerquery.2.tgz");let node=PathBuf::from(std::env::var_os("POST022_NODE").unwrap());let npm=PathBuf::from(std::env::var_os("POST022_NPM").unwrap());let runner=SystemNpmRunner::new(Some(home.clone()),root.join("npm-cache"));let probe=RealVersionProbe::default();let sink=NullProgressSink;let installer=RuntimeInstaller::new(&data,&runner,Some(home),&sink,&probe,Some(node.clone()));let req=|prefix:PathBuf,a:PathBuf|InstallRequest{prefix,source:InstallSource::Offline{archive:a},allow_scripts:false,npm_path:Some(npm.clone()),node_path:Some(node.clone())};
 let old=data.join("runtime/opencodex");let out=installer.install(&req(old.clone(),archive.clone())).expect("actual maintained archive offline install");assert_eq!(out.version,"2.97.0-post021.ownerquery.2");assert_eq!(probe.probe(None,&out.entry).as_deref(),Some(out.version.as_str()));println!("REAL_CHECK initial actual archive/npm/entry full version pass");
 let manifest=old.join(".runtime-manifest.json");let mut m:serde_json::Value=serde_json::from_slice(&std::fs::read(&manifest).unwrap()).unwrap();m["version"]=serde_json::json!("2.78.0");std::fs::write(&manifest,serde_json::to_vec(&m).unwrap()).unwrap();let store=RuntimeStore::new(&data);let mut record=store.load();record.resolved_version=Some("2.84.0".into());store.save(&record).unwrap();let previous=tree(&old);
 let config=data.join("opencodex-home/config.json");std::fs::create_dir_all(config.parent().unwrap()).unwrap();std::fs::write(&config,b"{\"providers\":{},\"sentinel\":true}").unwrap();let config_before=std::fs::read(&config).unwrap();assert!(installer.install(&req(old.clone(),archive.clone())).is_err());println!("REAL_CHECK mismatched original target blocked");
 let recovered=data.join("runtime/recovered");let next=installer.install(&req(recovered.clone(),archive.clone())).expect("actual new-prefix recovery");assert_eq!(tree(&old),previous);assert_eq!(std::fs::read(&config).unwrap(),config_before);let after=store.load();assert_eq!(after.source,RuntimeSourceKind::Managed);assert_eq!(after.installed_prefix(&data),recovered);assert_eq!(after.resolved_version.as_deref(),Some(next.version.as_str()));let package:serde_json::Value=serde_json::from_slice(&std::fs::read(recovered.join("node_modules/@bitkyc08/opencodex/package.json")).unwrap()).unwrap();let mf:serde_json::Value=serde_json::from_slice(&std::fs::read(recovered.join(".runtime-manifest.json")).unwrap()).unwrap();assert_eq!(package["version"],mf["version"]);assert_eq!(probe.probe(None,&next.entry).as_deref(),Some(next.version.as_str()));println!("REAL_CHECK recovery source/manifest/package/entry matched, old directory/config preserved");
 let record_before=std::fs::read(store.path()).unwrap();let entry_before=std::fs::read(&next.entry).unwrap();let bad=root.join("broken.tgz");std::fs::write(&bad,b"bad archive").unwrap();assert!(installer.install(&req(recovered.clone(),bad)).is_err());assert_eq!(std::fs::read(store.path()).unwrap(),record_before);assert_eq!(std::fs::read(&next.entry).unwrap(),entry_before);println!("REAL_CHECK invalid replacement keeps actual source and entry");
 let repeat=installer.install(&req(recovered.clone(),archive)).expect("later actual reinstall succeeds");assert_eq!(probe.probe(None,&repeat.entry).as_deref(),Some(repeat.version.as_str()));assert_eq!(tree(&old),previous);assert_eq!(std::fs::read(&config).unwrap(),config_before);println!("REAL_CHECK subsequent managed same-version reinstall pass; no OpenCodex/Codex service launched");
}
