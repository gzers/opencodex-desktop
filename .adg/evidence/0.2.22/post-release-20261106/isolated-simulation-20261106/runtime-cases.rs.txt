    fn post022_tree(path: &Path) -> Vec<(String, Vec<u8>)> {
        fn walk(root:&Path,p:&Path,out:&mut Vec<(String,Vec<u8>)>){
            if !p.exists(){return} for e in std::fs::read_dir(p).unwrap(){let p=e.unwrap().path();if p.is_dir(){walk(root,&p,out)}else{out.push((p.strip_prefix(root).unwrap().to_string_lossy().to_string(),std::fs::read(&p).unwrap()))}}
        }
        let mut out=vec![];walk(path,path,&mut out);out.sort();out
    }
